   105→function textStats(html: string): { words: number; chars: number } {
   106→  if (typeof window === 'undefined' || typeof DOMParser === 'undefined') {

   117→/** «#ff8800» یا «rgb(255,136,0)» → «FF8800» (بدون # برای docx) */
   118→function cssColorToHex(raw: string): string | null {
   119→  const v = raw.trim().toLowerCase()
   120→  if (!v || v === 'transparent' || v.startsWith('rgba(0, 0, 0, 0')) return null
   121→  if (v.startsWith('#')) {
   122→    const hex = v.slice(1)
   123→    if (hex.length === 3) return hex.split('').map((c) => c + c).join('').toUpperCase()
   124→    if (hex.length >= 6) return hex.slice(0, 6).toUpperCase()
   125→    return null
   126→  }
   127→  const m = v.match(/^rgba?\(\s*(\d+)\s*[,\s]\s*(\d+)\s*[,\s]\s*(\d+)/)
   128→  if (!m) return null
   129→  return [m[1], m[2], m[3]]
   130→    .map((n) => Math.max(0, Math.min(255, Number(n))))
   131→    .map((n) => n.toString(16).padStart(2, '0'))
   132→    .join('')
   133→    .toUpperCase()
   134→}

   138→/** docx فقط نام رنگ‌های استاندارد هایلایت را قبول می‌کند؛ نزدیک‌ترین را پیدا می‌کنیم */
   139→const HIGHLIGHT_SWATCHES: ReadonlyArray<readonly [string, string]> = [
   140→  ['yellow', 'FFFF00'], ['green', '00FF00'], ['cyan', '00FFFF'], ['magenta', 'FF00FF'],
   141→  ['red', 'FF0000'], ['blue', '0000FF'], ['darkYellow', '808000'], ['darkGreen', '008000'],
   142→  ['darkCyan', '008080'], ['darkMagenta', '800080'], ['darkRed', '800000'],
   143→  ['darkBlue', '000080'], ['darkGray', '808080'], ['lightGray', 'C0C0C0'],
   144→  ['black', '000000'], ['white', 'FFFFFF'],
   145→]

   147→function nearestHighlight(hex: string): DocxHighlight {
   148→  const r = parseInt(hex.slice(0, 2), 16)
   149→  const g = parseInt(hex.slice(2, 4), 16)
   150→  const b = parseInt(hex.slice(4, 6), 16)
   151→  let best = HIGHLIGHT_SWATCHES[0]
   152→  let bestDist = Number.POSITIVE_INFINITY
   153→  for (const sw of HIGHLIGHT_SWATCHES) {
   154→    const pr = parseInt(sw[1].slice(0, 2), 16)
   155→    const pg = parseInt(sw[1].slice(2, 4), 16)
   156→    const pb = parseInt(sw[1].slice(4, 6), 16)
   157→    const d = (r - pr) ** 2 + (g - pg) ** 2 + (b - pb) ** 2
   158→    if (d < bestDist) {
   159→      bestDist = d
   160→      best = sw
   161→    }
   162→  }
   163→  return best[0] as DocxHighlight
   164→}

   179→interface RunCtx {

   183→  strike: boolean

   185→  highlight?: DocxHighlight

   188→const BASE_CTX: RunCtx = { bold: false, italics: false, underline: false, strike: false }

   190→/** گره‌های درون‌خطی را به TextRun با ویژگی‌های bold/italic/underline/strike تبدیل می‌کند */
   191→function collectRuns(node: Node, ctx: RunCtx): TextRun[] {
   192→  if (node.nodeType === Node.TEXT_NODE) {
   193→    const text = node.textContent ?? ''
   194→    if (!text) return []
   195→    return [
   196→      new TextRun({
   197→        text,
   198→        bold: ctx.bold || undefined,
   199→        italics: ctx.italics || undefined,
   200→        underline: ctx.underline ? {} : undefined,
   201→        strike: ctx.strike || undefined,
   202→        color: ctx.color,
   203→        highlight: ctx.highlight,
   204→        rightToLeft: true,
   205→      }),
   206→    ]
   207→  }
   208→  if (node.nodeType !== Node.ELEMENT_NODE) return []
   209→  const el = node as HTMLElement
   210→  const tag = el.tagName
   211→  if (tag === 'BR') return [new TextRun({ break: 1, rightToLeft: true })]
   212→  if (tag === 'SCRIPT' || tag === 'STYLE') return []
   213→
   214→  const next: RunCtx = { ...ctx }
   215→  if (tag === 'B' || tag === 'STRONG') next.bold = true
   216→  if (tag === 'I' || tag === 'EM') next.italics = true
   217→  if (tag === 'U' || tag === 'INS') next.underline = true
   218→  if (tag === 'S' || tag === 'STRIKE' || tag === 'DEL') next.strike = true
   219→  if (tag === 'FONT') {
   220→    const c = cssColorToHex(el.getAttribute('color') ?? '')
   221→    if (c) next.color = c
   222→  }
   223→  const st = el.style
   224→  if (st) {
   225→    const w = st.fontWeight
   226→    if (w === 'bold' || w === 'bolder' || w === '700' || w === '800' || w === '900') next.bold = true
   227→    if (st.fontStyle === 'italic' || st.fontStyle === 'oblique') next.italics = true
   228→    const deco = st.textDecorationLine || st.textDecoration || ''
   229→    if (deco.includes('underline')) next.underline = true
   230→    if (deco.includes('line-through')) next.strike = true
   231→    const c = cssColorToHex(st.color)
   232→    if (c) next.color = c
   233→    const bg = cssColorToHex(st.backgroundColor)
   234→    if (bg) next.highlight = nearestHighlight(bg)
   235→  }
   236→
   237→  const runs: TextRun[] = []
   238→  for (const child of Array.from(el.childNodes)) runs.push(...collectRuns(child, next))
   239→  return runs
   240→}

   242→/** تراز پاراگراف — پیش‌فرض سند راست‌به‌چپ یعنی راست‌چین */
   243→function paraAlign(el: HTMLElement): (typeof AlignmentType)[keyof typeof AlignmentType] {
   244→  const a = (el.style.textAlign || el.getAttribute('align') || '').toLowerCase()
   245→  if (a === 'center') return AlignmentType.CENTER
   246→  if (a === 'left') return AlignmentType.LEFT
   247→  if (a === 'justify') return AlignmentType.JUSTIFIED
   248→  return AlignmentType.RIGHT
   249→}

   255→function buildDocxBody(root: HTMLElement): Paragraph[] {
   256→  const out: Paragraph[] = []
   257→  let olInstance = 0
   258→
   259→  const pushBlock = (el: HTMLElement) => {
   260→    const tag = el.tagName
   261→    const runs = collectRuns(el, BASE_CTX)
   262→    out.push(
   263→      new Paragraph({
   264→        heading:
   265→          tag === 'H1'
   266→            ? HeadingLevel.HEADING_1
   267→            : tag === 'H2'
   268→              ? HeadingLevel.HEADING_2
   269→              : tag === 'H3'
   270→                ? HeadingLevel.HEADING_3
   271→                : undefined,
   272→        bidirectional: true, // راست‌به‌چپ بودن پاراگراف (معادل RTL در docx نسخه ۹)
   273→        alignment: paraAlign(el),
   274→        children: runs.length > 0 ? runs : [emptyRun()],
   275→      })
   276→    )
   277→  }
   278→
   279→  const addList = (list: HTMLElement, ordered: boolean, level: number) => {
   280→    const instance = ordered ? ++olInstance : 0
   281→    for (const li of Array.from(list.children)) {
   282→      if (li.tagName !== 'LI') continue
   283→      const liEl = li as HTMLElement
   284→      // نسخهٔ مسطح از li بدون فهرست‌های تودرتو (آن‌ها جداگانه اضافه می‌شوند)
   285→      const flat = liEl.cloneNode(true) as HTMLElement
   286→      flat.querySelectorAll('ul,ol').forEach((n) => n.remove())
   287→      const runs = collectRuns(flat, BASE_CTX)
   288→      out.push(
   289→        new Paragraph({
   290→          ...(ordered
   291→            ? {
   292→                numbering: {
   293→                  reference: NUM_REF,
   294→                  level: Math.min(level, 2),
   295→                  instance,
   296→                },
   297→              }
   298→            : { bullet: { level: Math.min(level, 8) } }),
   299→          bidirectional: true,
   300→          alignment: paraAlign(liEl),
   301→          children: runs.length > 0 ? runs : [emptyRun()],
   302→        })
   303→      )
   304→      for (const sub of Array.from(liEl.children)) {
   305→        if (sub.tagName === 'UL') addList(sub as HTMLElement, false, level + 1)
   306→        else if (sub.tagName === 'OL') addList(sub as HTMLElement, true, level + 1)
   307→      }
   308→    }
   309→  }
   310→
   311→  const walk = (container: HTMLElement) => {
   312→    for (const child of Array.from(container.children)) {
   313→      const el = child as HTMLElement
   314→      switch (el.tagName) {
   315→        case 'H1':
   316→        case 'H2':
   317→        case 'H3':
   318→        case 'P':
   319→        case 'BLOCKQUOTE':
   320→          pushBlock(el)
   321→          break
   322→        case 'UL':
   323→          addList(el, false, 0)
   324→          break
   325→        case 'OL':
   326→          addList(el, true, 0)
   327→          break
   328→        case 'DIV':
   329→        case 'SECTION':
   330→        case 'ARTICLE':
   331→          if (el.querySelector('p,h1,h2,h3,ul,ol,div')) walk(el)
   332→          else pushBlock(el)
   333→          break
   334→        default:
   335→          pushBlock(el)
   336→      }
   337→    }
   338→  }
   339→
   340→  walk(root)
   341→  return out
   342→}

   510→      toast(
   511→        'ذخیره شد ✓',
   512→        `«${fileName}.txt» در پوشهٔ سندها ذخیره شد. حالا از «پرونده‌ها» در دسترس است.`,
   513→        CheckCircle2
   514→      )
   515→    } else {
   516→      toast('ذخیره نشد ✗', 'پوشهٔ سندها در فایل‌سیستم مجازی در دسترس نیست.', XCircle)
   517→    }
   518→  }
   519→
   520→  const handleOpen = () => {
   521→    const el = editorRef.current
   522→    if (!el) return
   523→    const fileName = safeName(name)
   524→    const node = getNode(useFs.getState().fs, `${DOCS_DIR}/${fileName}.txt`)
   525→    if (node && node.type === 'file') {
   526→      el.innerHTML = node.content || WELCOME_HTML
   527→      updateCounts()
   528→      refreshFmt()
   529→      toast('سند باز شد ✓', `«${fileName}.txt» از پوشهٔ سندها بارگذاری شد.`, FolderOpen)
   530→    } else {
   531→      toast('پیدا نشد ✗', `فایلی به نام «${fileName}.txt» در پوشهٔ سندها نیست.`, XCircle)
   532→    }
   533→  }
   534→
   535→  /* --------------------------- خروجی‌ها --------------------------- */
   536→
   537→  const exportDoc = () => {
   538→    const el = editorRef.current
   539→    if (!el) return
   540→    const fileName = safeName(name)
   541→    const stampHtml = stamp
   542→      ? `<p style="text-align:center;color:#9ca3af;font-size:10pt;margin-top:24pt">${STAMP_TEXT}</p>`
   543→      : ''
   544→    const html =
   545→      `<html xmlns:o="urn:schemas-microsoft-com:office:office" ` +
   546→      `xmlns:w="urn:schemas-microsoft-com:office:word" ` +
   547→      `xmlns="http://www.w3.org/TR/REC-html40" dir="rtl" lang="fa">` +
   548→      `<head><meta charset="utf-8" /><title>${escapeHtml(fileName)}</title></head>` +
   549→      `<body dir="rtl" style="font-family:Tahoma,'Vazirmatn',sans-serif;">` +
   550→      `${el.innerHTML}${stampHtml}</body></html>`
   551→    downloadBlob(new Blob(['\ufeff', html], { type: 'application/msword;charset=utf-8' }), `${fileName}.doc`)
   552→    toast('Word (.doc) آماده شد ✓', `«${fileName}.doc» دانلود شد.`, CheckCircle2)
   553→  }
   554→
   555→  const exportDocx = async () => {
   556→    const el = editorRef.current
   557→    if (!el) return
   558→    const fileName = safeName(name)
   559→    try {
   560→      const parsed = new DOMParser().parseFromString(el.innerHTML, 'text/html')
   561→      const children = buildDocxBody(parsed.body)
   562→      if (stamp) {
   563→        children.push(
   564→          new Paragraph({
   565→            alignment: AlignmentType.CENTER,
   566→            bidirectional: true,
   567→            spacing: { before: 480 },
   568→            children: [
   569→              new TextRun({ text: STAMP_TEXT, size: 16, color: '9CA3AF', rightToLeft: true }),
   570→            ],
   571→          })
   572→        )
   573→      }
   574→      const doc = new Document({
   575→        creator: 'نویسا — سان‌جاب‌او‌اس',
   576→        title: fileName,
   577→        numbering: {
   578→          config: [
   579→            {
   580→              reference: NUM_REF,
   581→              levels: [
   582→                { level: 0, format: LevelFormat.DECIMAL, text: '%1.' },
   583→                { level: 1, format: LevelFormat.LOWER_LETTER, text: '%2.' },
   584→                { level: 2, format: LevelFormat.LOWER_ROMAN, text: '%3.' },
   585→              ],
   586→            },
   587→          ],
   588→        },
   589→        sections: [
   590→          {
   591→            properties: {
   592→              page: { margin: { top: 1134, bottom: 1134, left: 1134, right: 1134 } },
   593→            },
   594→            children,
   595→          },
   596→        ],
   597→      })
   598→      const blob = await Packer.toBlob(doc)
   599→      downloadBlob(blob, `${fileName}.docx`)
   600→      toast('Word (.docx) آماده شد ✓', `«${fileName}.docx» دانلود شد.`, CheckCircle2)
   601→    } catch {
   602→      toast('خروجی docx شکست خورد', 'ساخت فایل Word استاندارد با خطا مواجه شد؛ خروجی .doc را امتحان کن.', XCircle)
   603→    }
   604→  }
   605→
   606→  const exportPdf = async () => {
   607→    const target = wrapRef.current
   608→    if (!target) return
   609→    const fileName = safeName(name)
   610→    try {
   611→      const canvas = await html2canvas(target, {
   612→        scale: 2,
   613→        backgroundColor: '#ffffff',
   614→        useCORS: true,
   615→      })
   616→      const pdf = new jsPDF({ orientation: 'portrait', unit: 'mm', format: 'a4' })
   617→      const pageW = 210 // میلی‌متر — A4
   618→      const pageH = 297
   619→      const imgH = (canvas.height * pageW) / canvas.width // ارتفاع کل سند بر حسب میلی‌متر
   620→      const data = canvas.toDataURL('image/jpeg', 0.95)
   621→      pdf.addImage(data, 'JPEG', 0, 0, pageW, imgH)
   622→      let rendered = pageH
   623→      while (rendered < imgH) {
   624→        pdf.addPage()
   625→        pdf.addImage(data, 'JPEG', 0, -rendered, pageW, imgH)
   626→        rendered += pageH
   627→      }
   628→      pdf.save(`${fileName}.pdf`)
   629→      toast('PDF آماده شد ✓', `«${fileName}.pdf» دانلود شد.`, CheckCircle2)
   630→    } catch {
   631→      toast('خروجی PDF شکست خورد', 'ساخت تصویر سند با خطا مواجه شد.', XCircle)
   632→    }
   633→  }
   634→
   635→  /* --------------------------- رابط کاربری --------------------------- */
   636→
   637→  return (
   638→    <div dir="rtl" className="flex h-full w-full flex-col bg-zinc-950/60 text-zinc-100">
   639→      {/* نوار ابزار */}
   640→      <header className="shrink-0 border-b border-white/10 bg-zinc-900/80">
   641→        <div className="flex flex-wrap items-center gap-0.5 px-2 py-1.5">
   642→          <FmtButton title="درشت (Ctrl+B)" active={fmt.bold} onAct={() => exec('bold')}>
   643→            <Bold className="size-4" />
   644→          </FmtButton>
   645→          <FmtButton title="کج (Ctrl+I)" active={fmt.italic} onAct={() => exec('italic')}>
   646→            <Italic className="size-4" />
   647→          </FmtButton>
   648→          <FmtButton title="زیرخط (Ctrl+U)" active={fmt.underline} onAct={() => exec('underline')}>
   649→            <UnderlineIcon className="size-4" />
   650→          </FmtButton>
   651→          <FmtButton title="خط‌خورده" active={fmt.strike} onAct={() => exec('strikeThrough')}>
   652→            <Strikethrough className="size-4" />
   653→          </FmtButton>
   654→          <Sep />
   655→          <FmtButton
   656→            title="پاراگراف"
   657→            active={fmt.block === 'p' || fmt.block === 'div'}
   658→            onAct={() => exec('formatBlock', '<p>')}
   659→          >
   660→            <Pilcrow className="size-4" />
   661→          </FmtButton>
   662→          <FmtButton title="تیتر ۱" active={fmt.block === 'h1'} onAct={() => exec('formatBlock', '<h1>')}>
   663→            <Heading1 className="size-4" />
   664→          </FmtButton>
   665→          <FmtButton title="تیتر ۲" active={fmt.block === 'h2'} onAct={() => exec('formatBlock', '<h2>')}>
   666→            <Heading2 className="size-4" />
   667→          </FmtButton>
   668→          <FmtButton title="تیتر ۳" active={fmt.block === 'h3'} onAct={() => exec('formatBlock', '<h3>')}>
   669→            <Heading3 className="size-4" />
   670→          </FmtButton>
   671→          <Sep />
   672→          <FmtButton title="فهرست نقطه‌ای" onAct={() => exec('insertUnorderedList')}>
   673→            <List className="size-4" />
   674→          </FmtButton>
   675→          <FmtButton title="فهرست شماره‌دار" onAct={() => exec('insertOrderedList')}>
   676→            <ListOrdered className="size-4" />
   677→          </FmtButton>
   678→          <Sep />
   679→          <FmtButton title="تراز از راست" onAct={() => exec('justifyRight')}>
   680→            <AlignRight className="size-4" />
   681→          </FmtButton>
   682→          <FmtButton title="تراز وسط" onAct={() => exec('justifyCenter')}>
   683→            <AlignCenter className="size-4" />
   684→          </FmtButton>
   685→          <FmtButton title="تراز از چپ" onAct={() => exec('justifyLeft')}>
   686→            <AlignLeft className="size-4" />
   687→          </FmtButton>
   688→          <Sep />
   689→          <FmtButton title="رنگ متن" onAct={() => colorRef.current?.click()}>
   690→            <Baseline className="size-4" />
   691→          </FmtButton>
   692→          <input
   693→            ref={colorRef}
   694→            type="color"
   695→            defaultValue="#111827"
   696→            tabIndex={-1}
   697→            aria-hidden
   698→            className="size-0 opacity-0"
   699→            onChange={(e) => exec('foreColor', e.target.value)}
   700→          />
   701→          <FmtButton title="رنگ هایلایت" onAct={() => hlRef.current?.click()}>
   702→            <Highlighter className="size-4" />
   703→          </FmtButton>
   704→          <input
   705→            ref={hlRef}
   706→            type="color"
   707→            defaultValue="#fde047"
   708→            tabIndex={-1}
   709→            aria-hidden
   710→            className="size-0 opacity-0"
   711→            onChange={(e) => exec('hiliteColor', e.target.value)}
   712→          />
   713→          <Sep />
   714→          <FmtButton title="واگرد (Ctrl+Z)" onAct={() => exec('undo')}>
   715→            <Undo2 className="size-4" />
   716→          </FmtButton>
   717→          <FmtButton title="ازنو (Ctrl+Y)" onAct={() => exec('redo')}>
   718→            <Redo2 className="size-4" />
   719→          </FmtButton>
   720→          <FmtButton title="پاک‌کردن قالب‌بندی" onAct={() => exec('removeFormat')}>
   721→            <Eraser className="size-4" />
   722→          </FmtButton>
   723→        </div>
   724→
   725→        {/* ردیف سند و خروجی‌ها */}
   726→        <div className="flex flex-wrap items-center gap-1.5 border-t border-white/5 px-2 py-1.5">
   727→          <input
   728→            value={name}
   729→            onChange={(e) => setName(e.target.value)}
   730→            placeholder="نام سند"
   731→            aria-label="نام سند"
   732→            className="h-8 w-32 rounded-lg border border-white/10 bg-white/5 px-2 text-xs text-zinc-100 outline-none placeholder:text-zinc-500 focus:border-white/25 sm:w-40"
   733→          />
   734→          <span className="hidden text-[10px] text-zinc-500 sm:inline">.txt</span>
   735→          <ActButton
   736→            title="ذخیره در پرونده‌ها"
   737→            hint="ذخیرهٔ سند در پوشهٔ سندهای فایل‌سیستم مجازی"
   738→            color={SUNJOB_PALETTE.teal}
   739→            icon={Save}
   740→            onClick={handleSave}
   741→          />
   742→          <ActButton
   743→            title="باز کردن آخرین سند"
   744→            hint="بازخوانی سند ذخیره‌شده از پوشهٔ سندها"
   745→            color="#a1a1aa"
   746→            icon={FolderOpen}
   747→            onClick={handleOpen}
   748→          />
   749→          <Sep />
   750→          <ActButton
   751→            title="Word (.docx)"
   752→            hint="خروجی Word استاندارد (docx واقعی)"
   753→            color={SUNJOB_PALETTE.violet}
   754→            icon={FileText}
   755→            onClick={() => void exportDocx()}
   756→          />
   757→          <ActButton
   758→            title="Word (.doc)"
   759→            hint="خروجی Word سازگار (سند HTML)"
   760→            color={SUNJOB_PALETTE.orange}
   761→            icon={File}
   762→            onClick={exportDoc}
   763→          />
   764→          <ActButton
   765→            title="PDF"
   766→            hint="خروجی PDF چندصفحه‌ای از سند"
   767→            color={SUNJOB_PALETTE.rose}
   768→            icon={FileDown}
   769→            onClick={() => void exportPdf()}
   770→          />
   771→          <ActButton
   772→            title="مهر سان‌جاب"
   773→            hint="درج یا حذف امضای اسپانسر پایین سند"
   774→            color={SUNJOB_PALETTE.tealBright}
   775→            icon={Stamp}
   776→            active={stamp}
   777→            onClick={() => setStamp((s) => !s)}
   778→          />
   779→        </div>
   780→      </header>
   781→
   782→      {/* صفحهٔ سند */}
   783→      <main className="os-scroll min-h-0 flex-1 overflow-y-auto p-3 sm:p-4">
   784→        <div
   785→          ref={wrapRef}
   786→          className="mx-auto w-full max-w-3xl rounded-xl bg-white text-zinc-900 shadow-xl ring-1 ring-black/10"
   787→        >
   788→          <div
   789→            ref={editorRef}
   790→            contentEditable
   791→            suppressContentEditableWarning
   792→            dir="rtl"
   793→            spellCheck={false}
   794→            role="textbox"
   795→            aria-multiline="true"
   796→            aria-label="متن سند"
   797→            onInput={handleInput}
   798→            onKeyUp={refreshFmt}
   799→            onMouseUp={refreshFmt}
   800→            onFocus={refreshFmt}
   801→            className={`os-scroll min-h-[420px] overflow-y-auto rounded-xl px-6 py-6 text-[15px] outline-none ${DOC_STYLES}`}
   802→            style={{ fontFamily: 'var(--font-vazir), sans-serif' }}
   803→          />
   804→          {/* مهر اسپانسر — بیرون از contentEditable ولی داخل ناحیهٔ خروجی PDF */}
   805→          {stamp && (
   806→            <div className="select-none pb-5 pt-1 text-center text-[11px] text-zinc-400">
   807→              {STAMP_TEXT}
   808→            </div>
   809→          )}
   810→        </div>
   811→      </main>
   812→
   813→      {/* نوار وضعیت */}
   814→      <footer className="flex shrink-0 flex-wrap items-center justify-between gap-2 border-t border-white/10 bg-zinc-900/80 px-3 py-1.5 text-[11px] text-zinc-400">
   815→        <div className="flex items-center gap-3">
   816→          <span>واژه‌ها: {faNum(words)}</span>
   817→          <span>نویسه‌ها: {faNum(chars)}</span>
   818→        </div>
   819→        <span className={autosave === 'saved' ? 'text-teal-400' : 'text-zinc-500'}>
   820→          {autosave === 'saving'
   821→            ? 'در حال ذخیرهٔ خودکار…'
   822→            : autosave === 'saved'
   823→              ? 'ذخیرهٔ خودکار انجام شد ✓'
   824→              : 'ذخیرهٔ خودکار فعال است'}
   825→        </span>
   826→      </footer>
   827→    </div>
   828→  )
   829→}
