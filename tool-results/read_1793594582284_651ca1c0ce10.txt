   188→   188→
   189→   189→function levenshtein(a: string, b: string): number {
   190→   190→  if (a === b) return 0
   191→   191→  if (!a.length) return b.length
   192→   192→  if (!b.length) return a.length
   193→   193→  let prev = Array.from({ length: b.length + 1 }, (_, i) => i)
   194→   194→  for (let i = 1; i <= a.length; i++) {
   195→   195→    const cur = [i]
   196→   196→    for (let j = 1; j <= b.length; j++) {
   197→   197→      cur[j] = Math.min(
   198→   198→        prev[j] + 1,
   199→   199→        cur[j - 1] + 1,
   200→   200→        prev[j - 1] + (a[i - 1] === b[j - 1] ? 0 : 1)
   201→   201→      )
   202→   202→    }
   203→   203→    prev = cur
   204→   204→  }
   205→   205→  return prev[b.length]
   206→   206→}

   215→   215→function didYouMean(input: string, extra: string[] = []): string | null {
   216→   216→  const pool = [...new Set([...ALL_COMMANDS, ...extra])]
   217→   217→  let best: string | null = null
   218→   218→  let bestD = Infinity
   219→   219→  for (const c of pool) {
   220→   220→    const d = levenshtein(input, c)
   221→   221→    if (d < bestD) {
   222→   222→      bestD = d
   223→   223→      best = c
   224→   224→    }
   225→   225→  }
   226→   226→  if (!best || bestD === 0) return null
   227→   227→  const maxD = input.length <= 3 ? 1 : Math.max(1, Math.floor(input.length / 3))
   228→   228→  return bestD <= maxD ? best : null
   229→   229→}

   231→   231→function nearestName(target: string, names: string[]): string | null {
   232→   232→  let best: string | null = null
   233→   233→  let bestD = Infinity
   234→   234→  for (const n of names) {
   235→   235→    const d = levenshtein(target, n)
   236→   236→    if (d < bestD) {
   237→   237→      bestD = d
   238→   238→      best = n
   239→   239→    }
   240→   240→  }
   241→   241→  return best && bestD <= 2 ? best : null
   242→   242→}

   286→   286→function banner(text: string): string {
   287→   287→  const chars = text.toUpperCase().slice(0, 12).split('')
   288→   288→  const rows = ['', '', '', '', '']
   289→   289→  for (const ch of chars) {
   290→   290→    const glyph = BANNER_FONT[ch] ?? BANNER_FONT[' ']
   291→   291→    for (let r = 0; r < 5; r++) rows[r] += glyph[r] + ' '
   292→   292→  }
   293→   293→  return rows.join('\n')
   294→   294→}
