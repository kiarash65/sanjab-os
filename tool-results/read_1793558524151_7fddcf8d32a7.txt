   189→function levenshtein(a: string, b: string): number {
   190→  if (a === b) return 0
   191→  if (!a.length) return b.length
   192→  if (!b.length) return a.length
   193→  let prev = Array.from({ length: b.length + 1 }, (_, i) => i)
   194→  for (let i = 1; i <= a.length; i++) {
   195→    const cur = [i]
   196→    for (let j = 1; j <= b.length; j++) {
   197→      cur[j] = Math.min(
   198→        prev[j] + 1,
   199→        cur[j - 1] + 1,
   200→        prev[j - 1] + (a[i - 1] === b[j - 1] ? 0 : 1)
   201→      )
   202→    }
   203→    prev = cur
   204→  }
   205→  return prev[b.length]
   206→}

   215→function didYouMean(input: string, extra: string[] = []): string | null {
   216→  const pool = [...new Set([...ALL_COMMANDS, ...extra])]
   217→  let best: string | null = null
   218→  let bestD = Infinity
   219→  for (const c of pool) {
   220→    const d = levenshtein(input, c)
   221→    if (d < bestD) {
   222→      bestD = d
   223→      best = c
   224→    }
   225→  }
   226→  if (!best || bestD === 0) return null
   227→  const maxD = input.length <= 3 ? 1 : Math.max(1, Math.floor(input.length / 3))
   228→  return bestD <= maxD ? best : null
   229→}

   231→function nearestName(target: string, names: string[]): string | null {
   232→  let best: string | null = null
   233→  let bestD = Infinity
   234→  for (const n of names) {
   235→    const d = levenshtein(target, n)
   236→    if (d < bestD) {
   237→      bestD = d
   238→      best = n
   239→    }
   240→  }
   241→  return best && bestD <= 2 ? best : null
   242→}

   286→function banner(text: string): string {
   287→  const chars = text.toUpperCase().slice(0, 12).split('')
   288→  const rows = ['', '', '', '', '']
   289→  for (const ch of chars) {
   290→    const glyph = BANNER_FONT[ch] ?? BANNER_FONT[' ']
   291→    for (let r = 0; r < 5; r++) rows[r] += glyph[r] + ' '
   292→  }
   293→  return rows.join('\n')
   294→}
