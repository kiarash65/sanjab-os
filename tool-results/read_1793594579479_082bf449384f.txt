   189→function levenshtein(a: string, b: string): number {
   190→  if (a === b) return 0
   191→  if (!a.length) return b.length
   192→  if (!b.length) return a.length
   193→  let prev = Array.from({ length: b.length + 1 }, (_, i) => i)
   194→  for (let i = 1; i <= a.length; i++) {
   195→    const cur = [i]
   196→    for (let j = 1; j <= b.length; j++) {
   197→      cur[j] = Math.min(
   198→        prev[j] + 1,
   199→        cur[j - 1] + 1,
   200→        prev[j - 1] + (a[i - 1] === b[j - 1] ? 0 : 1)
   201→      )
   202→    }
   203→    prev = cur
   204→  }
   205→  return prev[b.length]
   206→}

   215→function didYouMean(input: string, extra: string[] = []): string | null {
   216→  const pool = [...new Set([...ALL_COMMANDS, ...extra])]
   217→  let best: string | null = null
   218→  let bestD = Infinity
   219→  for (const c of pool) {
   220→    const d = levenshtein(input, c)
   221→    if (d < bestD) {
   222→      bestD = d
   223→      best = c
   224→    }
   225→  }
   226→  if (!best || bestD === 0) return null
   227→  const maxD = input.length <= 3 ? 1 : Math.max(1, Math.floor(input.length / 3))
   228→  return bestD <= maxD ? best : null
   229→}

   231→function nearestName(target: string, names: string[]): string | null {
   232→  let best: string | null = null
   233→  let bestD = Infinity
   234→  for (const n of names) {
   235→    const d = levenshtein(target, n)
   236→    if (d < bestD) {
   237→      bestD = d
   238→      best = n
   239→    }
   240→  }
   241→  return best && bestD <= 2 ? best : null
   242→}

   286→function banner(text: string): string {
   287→  const chars = text.toUpperCase().slice(0, 12).split('')
   288→  const rows = ['', '', '', '', '']
   289→  for (const ch of chars) {
   290→    const glyph = BANNER_FONT[ch] ?? BANNER_FONT[' ']
   291→    for (let r = 0; r < 5; r++) rows[r] += glyph[r] + ' '
   292→  }
   293→  return rows.join('\n')
   294→}

  1294→      n.startsWith(namePart)
  1295→    )
  1296→    if (names.length === 0) return
  1297→    if (names.length === 1) {
  1298→      const full = dirPart + names[0] + (node.children[names[0]].type === 'dir' ? '/' : '')
  1299→      parts[parts.length - 1] = full
  1300→      setInput(parts.join(' '))
  1301→    } else {
  1302→      push('dim', names.join('  '))
  1303→    }
  1304→  }
  1305→
  1306→  const onKeyDown = (e: ReactKeyboardEvent<HTMLInputElement>) => {
  1307→    if (e.ctrlKey && e.key.toLowerCase() === 'l') {
  1308→      e.preventDefault()
  1309→      setLines([])
  1310→      return
  1311→    }
  1312→    if (e.ctrlKey && e.key.toLowerCase() === 'c') {
  1313→      e.preventDefault()
  1314→      push('in', input + '^C')
  1315→      setInput('')
  1316→      return
  1317→    }
  1318→    if (e.key === 'Tab') {
  1319→      e.preventDefault()
  1320→      if (!input) return
  1321→      if (input.includes(' ')) {
  1322→        completePath()
  1323→      } else {
  1324→        const matches = ALL_COMMANDS.filter((c) => c.startsWith(input)).sort()
  1325→        if (matches.length === 1) {
  1326→          setInput(matches[0] + ' ')
  1327→        } else if (matches.length > 1) {
  1328→          push('dim', matches.join('  '))
  1329→        }
  1330→      }
  1331→      return
  1332→    }
  1333→    if (
  1334→      suggestion &&
  1335→      (e.key === 'Tab' ||
  1336→        (e.key === 'ArrowRight' &&
  1337→          (e.currentTarget.selectionStart ?? 0) >= input.length))
  1338→    ) {
  1339→      e.preventDefault()
  1340→      setInput(suggestion)
  1341→      return
  1342→    }
  1343→    if (e.key === 'ArrowUp') {
  1344→      e.preventDefault()
  1345→      if (history.length === 0) return
  1346→      const idx = Math.min(histIdx + 1, history.length - 1)
  1347→      setHistIdx(idx)
  1348→      setInput(history[idx])
  1349→    } else if (e.key === 'ArrowDown') {
  1350→      e.preventDefault()
  1351→      const idx = histIdx - 1
  1352→      if (idx < 0) {
  1353→        setHistIdx(-1)
  1354→        setInput('')
  1355→      } else {
  1356→        setHistIdx(idx)
  1357→        setInput(history[idx])
  1358→      }
  1359→    }
  1360→  }
  1361→
  1362→  const color: Record<LineType, string> = {
  1363→    in: 'text-zinc-500',
  1364→    out: 'text-zinc-200',
  1365→    ok: 'text-emerald-400',
  1366→    err: 'text-rose-400',
  1367→    warn: 'text-amber-300',
  1368→    sys: 'text-violet-300',
  1369→    dim: 'text-zinc-500',
  1370→  }
  1371→
  1372→  const prompt = `sanjab@sanjob-os:${prettyPath(cwd)}$`
  1373→
  1374→  return (
  1375→    <div
  1376→      className="relative flex h-full flex-col bg-[#0a0f0c] font-mono"
  1377→      onClick={() => inputRef.current?.focus()}
  1378→    >
  1379→      {/* نوار وضعیت tmux-وار */}
  1380→      <div
  1381→        className="flex shrink-0 items-center gap-2 border-b border-white/10 bg-zinc-900/80 px-3 py-1 text-[10px] text-zinc-400"
  1382→        dir="ltr"
  1383→      >
  1384→        <span className="font-black text-[var(--pengo-accent)]">pengo-sh 3.0</span>
  1385→        <span className="hidden truncate sm:inline">{prettyPath(cwd)}</span>
  1386→        <div className="flex-1" />
  1387→        <span
  1388→          className="flex items-center gap-1 rounded-full border border-white/10 bg-white/5 px-2 py-0.5"
  1389→          title="امتیاز آموزش ترمینال"
  1390→        >
  1391→          <Zap size={10} className="text-amber-300" aria-hidden />
  1392→          {tutor.xp} XP — {levelOf(tutor.xp)}
  1393→        </span>
  1394→        <span className="hidden md:inline">help | learn start</span>
  1395→      </div>
  1396→
  1397→      {/* خروجی */}
  1398→      <div
  1399→        ref={scrollRef}
  1400→        className="os-scroll min-h-0 flex-1 overflow-y-auto p-3 text-[12px] leading-6 sm:text-[13px]"
  1401→        dir="rtl"
  1402→      >
  1403→        {lines.map((l) =>
  1404→          l.type === 'in' ? (
  1405→            <pre key={l.id} className="whitespace-pre-wrap text-left" dir="ltr">
  1406→              <span className="text-emerald-500">{prompt} </span>
  1407→              <span className="text-zinc-200">{l.text}</span>
  1408→            </pre>
  1409→          ) : l.text.includes('\n') || /^[\s/\]|<>_()=\\#▓░▏▔▕├└─]+$/.test(l.text) ? (
  1410→            <pre
  1411→              key={l.id}
  1412→              className={`whitespace-pre text-left ${color[l.type]}`}
  1413→              dir="ltr"
  1414→            >
  1415→              {l.text}
  1416→            </pre>
  1417→          ) : (
  1418→            <div key={l.id} className={`whitespace-pre-wrap ${color[l.type]}`}>
  1419→              {l.text}
  1420→            </div>
  1421→          )
  1422→        )}
  1423→      </div>
  1424→
  1425→      {/* نوار پیشنهاد اصلاح غلط */}
  1426→      {activeFix && (
  1427→        <div
  1428→          className="pengo-wiggle flex shrink-0 flex-wrap items-center gap-2 border-t border-amber-400/20 bg-amber-500/10 px-3 py-1.5 text-[11px]"
  1429→          dir="rtl"
  1430→        >
  1431→          <Sparkles size={13} className="shrink-0 text-amber-300" aria-hidden />
  1432→          <span className="text-amber-100">منظورت این بود؟</span>
  1433→          <code
  1434→            className="rounded bg-black/50 px-1.5 py-0.5 font-bold text-amber-200"
  1435→            dir="ltr"

  1437→            {activeFix.fixed}

  1471→          <input

  1483→          {suggestion && suggestion.startsWith(input) && (
